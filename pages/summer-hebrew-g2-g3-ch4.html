הַמִּלָּה הַחֲסֵרָה מְחַכָּה בַּמִּשְׁפָּט

כְּשֶׁחֲסֵרָה מִלָּה בְּמִשְׁפָּט, אֵין צֹרֶךְ לְנַחֵשׁ. הַמִּשְׁפָּט עַצְמוֹ אוֹמֵר לָנוּ מָה מַתְאִים לוֹ, וְצָרִיךְ רַק לִקְרֹא אוֹתוֹ עַד הַסּוֹף לִפְנֵי שֶׁמַּחְלִיטִים. בַּפֶּרֶק הַזֶּה נִלְמַד שְׁלוֹשָׁה דְּבָרִים שֶׁהוֹלְכִים יַחַד: לְהַשְׁלִים מִלָּה לְפִי מָה שֶׁסְּבִיבָהּ; לְהַכִּיר מִלִּים שֶׁהוֹלְכוֹת יַחַד בְּעִבְרִית, וּבִטּוּיִים שֶׁאוֹמְרִים מַשֶּׁהוּ אַחֵר מִמָּה שֶׁכָּתוּב בָּהֶם; וְלִקְרֹא קֶטַע שָׁלֵם וְלָדַעַת אֵיפֹה בְּדִיּוּק מְחַכָּה הַתְּשׁוּבָה. לַקֶּטַע שֶׁבַּדַּף יֵשׁ מִלּוֹן קָטָן בַּצַּד, וְהוּא לֹא בַּסּוֹף.
בַּפֶּרֶק 18 שְׁאֵלוֹת בִּשְׁלוֹשָׁה מַעֲרָכִים, וְעוֹד חָמֵשׁ מְשִׂימוֹת פְּתוּחוֹת.
חֲזָרָה לְשַׁעַר הַחוֹבֶרֶת

מַשְׁלִימִים וְקוֹרְאִים בַּחֲרוּ מִלָּה, וְקִבְּלוּ גַּם הֶסְבֵּר לָמָּה הָאֲחֵרוֹת אֵינָן מַתְאִימוֹת
קִרְאוּ אֶת הַמִּשְׁפָּט עַד הַסּוֹף
וְרַק אַחַר כָּךְ בַּחֲרוּ אֶת הַמִּלָּה שֶׁחֲסֵרָה בּוֹ
מַתְאִימָה אֵינָהּ מַתְאִימָה

יֵשׁ מִלִּים שֶׁהוֹלְכוֹת תָּמִיד יַחַד. כְּשֶׁשְּׁתֵּי מִלִּים נִצְמָדוֹת זוֹ לְזוֹ מַסְפִּיק פְּעָמִים רַבּוֹת, הֵן מַפְסִיקוֹת לִהְיוֹת שְׁתַּיִם וְהוֹפְכוֹת לְדָבָר אֶחָד עִם מַשְׁמָעוּת מִשֶּׁלּוֹ. הָעַיִן לוֹמֶדֶת אֶת הַצֶּמֶד וְתוֹפֶסֶת אוֹתוֹ בְּבַת אַחַת, וְלֹא מִלָּה אַחֲרֵי מִלָּה. זֶה בְּדִיּוּק מָה שֶׁעוֹשֶׂה אֶת הַקְּרִיאָה לִמְהִירָה יוֹתֵר.

קִרְאוּ קֹדֶם, וְהַמִּלּוֹן מְחַכֶּה בַּצַּד

לִפְנֵיכֶם קֶטַע קָצָר בֶּן שְׁמוֹנֶה שׁוּרוֹת, וְכָל שׁוּרָה מְמֻסְפֶּרֶת כְּדֵי שֶׁתּוּכְלוּ לְהַצְבִּיעַ עַל הַמָּקוֹם שֶׁבּוֹ מְחַכָּה הַתְּשׁוּבָה. חָמֵשׁ מִלִּים בַּקֶּטַע מְסֻמָּנוֹת, וּלְכָל אַחַת מֵהֶן יֵשׁ הֶסְבֵּר קָצָר בַּצַּד, בְּדִיּוּק לְיַד הַשּׁוּרָה. זֶה נִקְרָא מִלּוֹן שׁוּלַיִם, וְהוּא נוֹלַד כְּדֵי שֶׁלֹּא תִּצְטָרְכוּ לַעֲצֹר אֶת הַקְּרִיאָה וּלְחַפֵּשׂ בְּסוֹף הַדַּף. קִרְאוּ אֶת הַקֶּטַע פַּעַם אַחַת בְּלִי לְהִתְעַכֵּב, וְאַחַר כָּךְ עוֹד פַּעַם עִם הַמִּלּוֹן. וְעוֹד דָּבָר: לַקֶּטַע אֵין כּוֹתֶרֶת, וְזֶה בְּכַוָּנָה. אֶחָד הַתַּרְגִּילִים בְּהֶמְשֵׁךְ יְבַקֵּשׁ מִכֶּם לִבְחֹר לוֹ אַחַת.

שְׁמוֹנֶה שׁוּרוֹת · חָמֵשׁ מִלִּים בַּמִּלּוֹן שֶׁבַּצַּד
קֶטַע קְרִיאָה שְׁמוֹנֶה שׁוּרוֹת מְמֻסְפָּרוֹת · הַכּוֹתֶרֶת חֲסֵרָה בְּכַוָּנָה
1בַּסִּפְרִיָּה הַשְּׁכוּנָתִית גָּר חָתוּל כָּתֹם, וּשְׁמוֹ שׁוּקוֹ.
2שׁוּקוֹ הִגִּיעַ לְשָׁם בַּחֹרֶף, רָטֹב וְרָעֵב, וְהַסַּפְרָנִית פָּתְחָה לוֹ אֶת הַדֶּלֶת.
3מֵאָז הוּא יָשֵׁן כָּל בֹּקֶר עַל הַמַּדָּף הַתַּחְתּוֹן, בֵּין סִפְרֵי הַחַיּוֹת.
4בְּשָׁעָה אַרְבַּע מַגִּיעִים הַיְלָדִים, וְשׁוּקוֹ מִתְעוֹרֵר וּמְקַבֵּל אֶת פְּנֵיהֶם.
5יֵשׁ שָׁם יֶלֶד אֶחָד, יוֹתָם, וְהוּא מִתְבַּיֵּשׁ לִקְרֹא בְּקוֹל לִפְנֵי הַכִּתָּה.
6אֲבָל לְיַד שׁוּקוֹ הוּא קוֹרֵא בְּלִי לְהַסְמִיק, מִפְּנֵי שֶׁחָתוּל אֵינוֹ מְתַקֵּן וְאֵינוֹ צוֹחֵק.
7הַסַּפְרָנִית אוֹמֶרֶת שֶׁמֵּאָז שׁוּקוֹ הִגִּיעַ, הַיְלָדִים מַשְׁאִילִים יוֹתֵר סְפָרִים.
8וְיוֹתָם? הוּא כְּבָר בָּלַע שְׁלוֹשָׁה סְפָרִים בְּחֹדֶשׁ אֶחָד.
מִלּוֹן קָטָן בַּצַּד
שְׁכוּנָתִית שֶׁל הַשְּׁכוּנָה, הַמָּקוֹם שֶׁגָּרִים בּוֹ.
מְקַבֵּל אֶת פְּנֵיהֶם יוֹצֵא לִקְרָאתָם וְשָׂמֵחַ שֶׁבָּאוּ.
לְהַסְמִיק לְהִתְאַדֵּם בַּפָּנִים מֵרֹב בּוּשָׁה.
מַשְׁאִילִים לוֹקְחִים מַשֶּׁהוּ לִזְמַן מָה, וּמַחֲזִירִים אוֹתוֹ אַחַר כָּךְ.
בָּלַע סְפָרִים קָרָא אוֹתָם מַהֵר וּבַהֲנָאָה. אַף אֶחָד לֹא בָּלַע כְּלוּם בֶּאֱמֶת.
חֲמֵשׁ הַמִּלִּים הַמְּסֻמָּנוֹת מְחַכּוֹת לָכֶם בַּצַּד, וְלֹא בְּסוֹף הַחוֹבֶרֶת. כְּשֶׁנִּתְקָלִים בְּמִלָּה שֶׁאֵינָהּ מֻכֶּרֶת, קוֹרְאִים אֶת הַהֶסְבֵּר הַקָּצָר וְחוֹזְרִים מִיָּד אֶל הַשּׁוּרָה. וְשִׂימוּ לֵב לַמִּלָּה הָאַחֲרוֹנָה בַּמִּלּוֹן: הִיא אֵינָהּ אוֹמֶרֶת מָה שֶׁכָּתוּב בָּהּ.
שֵׁשׁ תִּזְכֹּרוֹת לִפְנֵי הַתִּרְגּוּל

מָה חָשׁוּב לִזְכֹּר

שָׁלוֹשׁ כַּרְטִיסִיּוֹת עַל הַשְׁלָמַת מִלָּה וְעַל צֵרוּפִים, וְשָׁלוֹשׁ עַל קְרִיאַת קֶטַע שָׁלֵם.

1

קוֹרְאִים עַד הַסּוֹף

הַטָּעוּת הַשְּׁכִיחָה בְּיוֹתֵר בְּמִשְׁפַּט הַשְׁלָמָה הִיא לְהַחְלִיט בָּאֶמְצַע. לִפְעָמִים דַּוְקָא הַמִּלִּים שֶׁאַחֲרֵי הַמָּקוֹם הָרֵיק הֵן שֶׁמַּכְרִיעוֹת. לָכֵן קוֹרְאִים אֶת הַמִּשְׁפָּט כֻּלּוֹ, וְרַק אָז בּוֹחֲרִים.

2

מִלִּים שֶׁהוֹלְכוֹת יַחַד

צֵרוּף הוּא שְׁתֵּי מִלִּים שֶׁיַּחַד אוֹמְרוֹת דָּבָר אֶחָד. בֵּית סֵפֶר אֵינוֹ בַּיִת וְאֵינוֹ סֵפֶר, וּמֶזֶג אֲוִיר אֵינוֹ שְׁנֵי דְּבָרִים אֶלָּא אֶחָד: אִם חַם אוֹ קַר בַּחוּץ.

3

הַפֹּעַל שֶׁהוֹלֵם אֶת הַשֵּׁם

לֹא כָּל פֹּעַל הוֹלֵךְ עִם כָּל שֵׁם, גַּם כְּשֶׁהַמַּשְׁמָעוּת קְרוֹבָה. שׁוֹתִים מַיִם, וְאֵין אוֹכְלִים מַיִם. חוֹצִים כְּבִישׁ בְּמַעֲבַר חֲצִיָּה, וְאֵין עוֹבְרִים אוֹתוֹ בִּשְׂחִיָּה. הָאֹזֶן מַרְגִּישָׁה זֹאת לִפְנֵי הָעַיִן.

4

בִּטּוּי שֶׁאוֹמֵר מַשֶּׁהוּ אַחֵר

יֵשׁ צֵרוּפִים שֶׁמַּשְׁמָעוּתָם אֵינָהּ סְכוּם הַמִּלִּים שֶׁבָּהֶם. לֵב שֶׁל אֶבֶן אֵינוֹ לֵב וְאֵינוֹ אֶבֶן: כָּךְ אוֹמְרִים עַל מִי שֶׁאֵינוֹ מְרַחֵם. בִּטּוּי כָּזֶה מְפָרְשִׁים כְּיָחִידָה אַחַת, וְלֹא מִלָּה מִלָּה.

5

מִלָּה חֲדָשָׁה בְּתוֹךְ שׁוּרָה

כְּשֶׁפּוֹגְשִׁים מִלָּה שֶׁאֵינָהּ מֻכֶּרֶת, לֹא עוֹצְרִים מִיָּד. קוֹרְאִים אֶת הַמִּשְׁפָּט שֶׁלִּפְנֵי וְאֶת הַמִּשְׁפָּט שֶׁאַחֲרֵי, וּמְשַׁעֲרִים. וְאִם יֵשׁ מִלּוֹן בַּצַּד, זוֹ בְּדִיּוּק הַשְּׁנִיָּה לְהִסְתַּכֵּל בּוֹ.

6

אֵיפֹה מְחַכָּה הַתְּשׁוּבָה

יֵשׁ שְׁאֵלוֹת שֶׁהַתְּשׁוּבָה שֶׁלָּהֶן כְּתוּבָה בַּקֶּטַע, וְאָז מְחַפְּשִׂים אֶת הַשּׁוּרָה וּמַצְבִּיעִים עָלֶיהָ. וְיֵשׁ שְׁאֵלוֹת שֶׁהַתְּשׁוּבָה שֶׁלָּהֶן אֵינָהּ כְּתוּבָה, וְאֶפְשָׁר לְהָבִין אוֹתָהּ מִן הַכָּתוּב. לָדַעַת בְּאֵיזוֹ שְׁאֵלָה אַתֶּם עוֹמְדִים חוֹסֵךְ חֲצִי מִן הַזְּמַן.

מְקַבְּלִים הַחְלָטָה
כָּךְ אוֹמְרִים בְּעִבְרִית
זֶה הַצֵּרוּף שֶׁאוֹמְרִים בְּעִבְרִית. הוּא נִשְׁמָע רָגִיל, וְאִישׁ אֵינוֹ עוֹצֵר עָלָיו.
לוֹקְחִים הַחְלָטָה
צוֹרֵם, וְאֵין אוֹמְרִים כָּךְ
כָּל מִלָּה כָּאן תְּקִינָה בִּפְנֵי עַצְמָהּ, וְהַצֵּרוּף בְּכָל זֹאת צוֹרֵם. הָאֹזֶן תּוֹפֶסֶת זֹאת לִפְנֵי הַכְּלָלִים.
נִפְתֹּר יַחַד, צַעַד אַחַר צַעַד

שָׁלוֹשׁ דֻּגְמָאוֹת פְּתוּרוֹת
